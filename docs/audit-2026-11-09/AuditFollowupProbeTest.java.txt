import org.junit.jupiter.api.Test;
import org.mockito.ArgumentCaptor;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.crypto.password.PasswordEncoder;
import java.math.BigDecimal;
import java.time.LocalDateTime;
import java.time.LocalDate;
import java.util.*;
import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.Mockito.*;
import static org.mockito.ArgumentMatchers.*;

// Audit probes record faulty behavior; keep outside the regular regression suite.
class AuditFollowupProbeTest {
    private UsernamePasswordAuthenticationToken actor(String role) {
        return new UsernamePasswordAuthenticationToken("9", "", List.of(new SimpleGrantedAuthority("ROLE_" + role)));
    }

    @Test void memberCanCreateAiPlanForAnotherMember() {
        WorkoutPlanRepository plans = mock(WorkoutPlanRepository.class);
        when(plans.save(any())).thenAnswer(i -> i.getArgument(0));
        WorkoutProgressController c = new WorkoutProgressController(plans,
            mock(BodyProgressLogRepository.class), mock(UserRepository.class), mock(EngagementService.class));
        WorkoutPlan saved = c.createPlan(new WorkoutProgressController.PlanRequest(
            777L, "Audit", null, "[]", null, null, null, "AI"), actor("MEMBER"));
        assertEquals(777L, saved.getMemberId());
        verify(plans).save(any());
    }

    @Test void unrelatedCoachCanReadAnyMembersWorkoutHistory() {
        WorkoutPlanRepository plans = mock(WorkoutPlanRepository.class);
        BodyProgressLogRepository logs = mock(BodyProgressLogRepository.class);
        UserRepository users = mock(UserRepository.class);
        when(users.findById(777L)).thenReturn(Optional.of(User.builder().userId(777L).fullName("Audit").build()));
        when(plans.findByMemberIdOrderByCreatedAtDesc(777L)).thenReturn(List.of());
        when(logs.findByMemberIdOrderByRecordedDateAsc(777L)).thenReturn(List.of());
        WorkoutProgressController c = new WorkoutProgressController(plans, logs, users, mock(EngagementService.class));
        assertEquals(777L, c.memberWorkoutProgress(777L, actor("COACH")).get("memberId"));
        verify(logs).findByMemberIdOrderByRecordedDateAsc(777L);
    }

    @Test void pendingStaffCanActivateWithResentOtpWithoutManagerApproval() {
        UserRepository users = mock(UserRepository.class);
        AuthSessionRepository sessions = mock(AuthSessionRepository.class);
        VerificationTokenRepository tokens = mock(VerificationTokenRepository.class);
        EmailService email = mock(EmailService.class);
        User staff = User.builder().userId(9L).email("audit@example.test").fullName("Audit")
            .role("RECEPTIONIST").status("PENDING").build();
        when(users.findByEmailIgnoreCase(staff.getEmail())).thenReturn(Optional.of(staff));
        when(users.save(any())).thenAnswer(i -> i.getArgument(0));
        when(tokens.findByEmailAndTokenTypeAndUsedAtIsNullAndExpiresAtAfter(anyString(), anyString(), any()))
            .thenReturn(Optional.empty());
        AuthService service = new AuthService(users, sessions, tokens, mock(PasswordEncoder.class), email, mock(AuditLogService.class));
        service.resendOtp(new ResendOtpRequest(staff.getEmail()));
        ArgumentCaptor<String> otp = ArgumentCaptor.forClass(String.class);
        verify(email).sendRegistrationOtpEmail(eq(staff.getEmail()), eq(staff.getFullName()), otp.capture());
        ArgumentCaptor<VerificationToken> token = ArgumentCaptor.forClass(VerificationToken.class);
        verify(tokens).save(token.capture());
        token.getValue().setCreatedAt(LocalDateTime.now());
        when(tokens.findByEmailAndTokenTypeAndUsedAtIsNullAndExpiresAtAfter(anyString(), anyString(), any()))
            .thenReturn(Optional.of(token.getValue()));
        AuthResponse response = service.verifyEmail(new VerifyOtpRequest(staff.getEmail(), otp.getValue()));
        assertEquals("ACTIVE", staff.getStatus());
        assertEquals("RECEPTIONIST", response.getRole());
        assertNotNull(response.getToken());
        verify(sessions).save(any());
    }

    @Test void replayedSuccessfulCallbackChangesRefundedPaymentBackToPaid() {
        VNPayGateway gateway = mock(VNPayGateway.class);
        PaymentTransactionRepository payments = mock(PaymentTransactionRepository.class);
        PaymentTransaction payment = new PaymentTransaction();
        payment.setPaymentId(2L); payment.setAmount(BigDecimal.TEN); payment.setPaymentStatus("REFUNDED");
        when(payments.findById(2L)).thenReturn(Optional.of(payment));
        when(gateway.verifySignature(anyMap(), anyString())).thenReturn(true);
        VNPayWebhookController c = new VNPayWebhookController(gateway, payments, List.of());
        c.handleIPN(Map.of("vnp_SecureHash", "mock-valid", "vnp_TxnRef", "2_123", "vnp_Amount", "1000",
            "vnp_ResponseCode", "00", "vnp_TransactionNo", "audit-txn", "vnp_PayDate", "20261109120000"));
        assertEquals("PAID", payment.getPaymentStatus());
        verify(payments).save(payment);
    }

    @Test void leaveApprovalDoubleBooksReplacementCoach() {
        CoachLeaveRequestRepository leaves = mock(CoachLeaveRequestRepository.class);
        UserRepository users = mock(UserRepository.class);
        ClassSessionRepository classes = mock(ClassSessionRepository.class);
        LocalDate day = LocalDate.now().plusDays(1);
        CoachLeaveRequest leave = CoachLeaveRequest.builder().requestId(1L).coachId(10L)
            .startDate(day).endDate(day).status("PENDING").build();
        ClassSession original = new ClassSession();
        original.setSessionId(1L); original.setCoachId(10L); original.setStatus("SCHEDULED");
        original.setStartTime(day.atTime(10, 0)); original.setEndTime(day.atTime(11, 0));
        ClassSession conflict = new ClassSession();
        conflict.setSessionId(2L); conflict.setCoachId(20L); conflict.setStatus("SCHEDULED");
        conflict.setStartTime(day.atTime(10, 0)); conflict.setEndTime(day.atTime(11, 0));
        when(leaves.findById(1L)).thenReturn(Optional.of(leave));
        when(leaves.save(any())).thenAnswer(i -> i.getArgument(0));
        when(users.findById(20L)).thenReturn(Optional.of(User.builder().userId(20L).role("COACH").status("ACTIVE").build()));
        when(classes.findAll()).thenReturn(List.of(original, conflict));
        CoachLeaveController c = new CoachLeaveController(leaves, users, classes, mock(BookingRepository.class));
        c.decideLeaveRequest(1L, new LeaveDecisionRequest(true, 20L, "Audit"), actor("MANAGER"));
        assertEquals("APPROVED", leave.getStatus());
        assertEquals(conflict.getCoachId(), original.getCoachId());
        verify(classes).saveAll(anyList());
    }

    @Test void paymentStatusExposesAnotherMembersTransactionWithoutOwnershipCheck() {
        PaymentTransactionRepository payments = mock(PaymentTransactionRepository.class);
        PaymentTransaction payment = new PaymentTransaction();
        payment.setPaymentId(2L); payment.setMemberId(777L); payment.setAmount(BigDecimal.TEN);
        when(payments.findById(2L)).thenReturn(Optional.of(payment));
        PaymentController c = new PaymentController(payments, mock(VNPayGateway.class));
        assertEquals(200, c.getPaymentStatus(2L).getStatusCode().value());
        verify(payments).findById(2L);
    }

    @Test void unpaidPtIsConfirmedAndExcludedFromTimeoutScan() {
        BookingRepository bookings = mock(BookingRepository.class);
        ClassSessionRepository classes = mock(ClassSessionRepository.class);
        UserRepository users = mock(UserRepository.class);
        CourtBookingRepository courts = mock(CourtBookingRepository.class);
        PaymentTransactionRepository payments = mock(PaymentTransactionRepository.class);
        PaymentService paymentService = mock(PaymentService.class);
        when(users.findById(9L)).thenReturn(Optional.of(User.builder().userId(9L).role("MEMBER").status("ACTIVE").build()));
        when(users.findById(20L)).thenReturn(Optional.of(User.builder().userId(20L).role("COACH").fullName("Audit").build()));
        PaymentTransaction payment = new PaymentTransaction();
        payment.setPaymentId(2L); payment.setAmount(BigDecimal.valueOf(250000)); payment.setPaymentStatus("PENDING");
        when(paymentService.createPayment(eq(9L), eq("PT"), nullable(String.class), any(), eq("VNPAY"))).thenReturn(payment);
        when(classes.save(any())).thenAnswer(i -> { ClassSession s = i.getArgument(0); s.setSessionId(3L); return s; });
        when(bookings.save(any())).thenAnswer(i -> { Booking b = i.getArgument(0); b.setBookingId(4L); return b; });
        BookingService service = new BookingService(bookings, classes, users, mock(PublicSportRepository.class),
            mock(PublicFacilityRepository.class), mock(SportsCourtRepository.class), courts,
            mock(CourtMaintenanceRepository.class), payments, mock(UserMembershipRepository.class),
            mock(MembershipPackageRepository.class), mock(MembershipService.class), mock(AuditLogService.class), paymentService);
        LocalDateTime start = LocalDateTime.now().plusDays(1);
        service.checkout(9L, new CheckoutRequest("PT", 20L, start, start.plusHours(1), "VNPAY"));
        ArgumentCaptor<Booking> saved = ArgumentCaptor.forClass(Booking.class);
        verify(bookings).save(saved.capture());
        assertEquals("CONFIRMED", saved.getValue().getStatus());
        assertEquals("PENDING", payment.getPaymentStatus());
        new PaymentTimeoutScheduler(bookings, courts, payments, null).cancelUnpaidBookings();
        verify(bookings).findByStatusIgnoreCaseAndBookingTimeBefore(eq("BOOKED"), any());
        verify(bookings, never()).findByStatusIgnoreCaseAndBookingTimeBefore(eq("CONFIRMED"), any());
    }
}
