package com.sporthub.backend;

import org.junit.jupiter.api.Test;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.transaction.annotation.Transactional;
import java.time.LocalDateTime;

@SpringBootTest(properties = "sporthub.scheduling.enabled=false")
@Transactional(readOnly = true)
class BackgroundAuditReadbackTest {
    @Autowired org.springframework.jdbc.core.JdbcTemplate jdbc;
    @Test void readBackgroundJobCountsDuringTheInterruptedDiagnosticRun() {
        var counts = jdbc.queryForList("select action, count(*) as event_count from audit_logs where actor_role = 'SYSTEM' and created_at >= ? and created_at < ? group by action",
            LocalDateTime.of(2026, 11, 9, 18, 31, 53), LocalDateTime.of(2026, 11, 9, 18, 38, 0));
        System.out.println("BACKGROUND_JOB_AUDIT_COUNTS=" + counts);
    }
}
